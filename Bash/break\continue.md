# break and continue
### break и continue — это команды, которые управляют поведением изнутри уже работающего цикла (for или while), меняя его обычный ход выполнения

---

# break — досрочно выйти из цикла целиком
### Обычно цикл заканчивается сам, когда список закончился (for) или условие стало ложным (while). break позволяет прервать цикл раньше времени, даже если формально он мог бы продолжаться
```bash
#!/bin/bash
for ip in 192.168.3.1 192.168.3.109 192.168.3.116 192.168.3.200
do
    echo "Проверяю $ip"
    if [ "$ip" = "192.168.3.116" ]; then
        echo "Нашел нужный хост, останавливаюсь"
        break
    fi
done

Как только ip станет равен 192.168.3.116, выполнится break — и цикл сразу прекратится, даже не дойдя до 192.168.3.200, который формально ещё оставался в списке
```

---

# continue — пропустить текущий шаг, но продолжить цикл дальше
### В отличие от break (полная остановка), continue просто прерывает текущую итерацию и сразу переходит к следующему элементу, не выполняя остаток кода внутри этой итерации
```bash
#!/bin/bash
for ip in 192.168.3.1 192.168.3.109 192.168.3.116
do
    if [ "$ip" = "192.168.3.109" ];then
    echo "Пропускаю $ip"
    continue
    fi
    echo "Сканирую $ip"
    ping -c1 "$ip" > /dev/null
done

Тут для 192.168.3.109 выполнится только echo "Пропускаю...", а строки echo "Сканирую..." и ping для этого конкретного IP не выполнятся — continue перескочит сразу к следующему элементу списка (192.168.3.116). Остальные IP обработаются полностью, как обычно
```

# Общий пример на оба оператора
```bash
file=$1

for ip in $(cat "$file")
do 
    if [ "$ip" = "192.168.3.109" ]; then
        echo "$ip это моя виртуалка ее не трогать!!!"
        continue
    elif [ "$ip" = "192.168.3.116" ]; then
        echo "$ip это тоже моя виртуалка, не трогать"
        continue
    elif [ "$ip" = "192.168.3.1" ]; then
        echo "$ip это роутер, не трогать"
        continue
    fi

    ping -c1 "$ip" > /dev/null
    if [ $? -ne 0 ]; then
        echo "Обнаружен недоступный хост: $ip, прекаращаю проверку"
    fi
    break
done

continue действует только в той конкретной точке кода, где он написан — он не "запоминает", что где-то раньше уже сработал похожий пропуск. Каждая ветка if/elif, которая должна пропускать итерацию, обязана сама содержать свой собственный continue — bash не переносит эту логику автоматически между ветками, даже если они выглядят структурно похожими
```
