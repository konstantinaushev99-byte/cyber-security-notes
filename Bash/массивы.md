# Массивы
Что это - обычная переменная хранит одно значение:
```bash
ip="192.168.3.109"
```
массив хранит несколько значений под одни м именем:
```bash
ips=("192.168.3.1" "192.168.3.109" "192.168.3.116")
```
Также элементы массива нумеруются с нуля:
```bash
echo "${ips[0]}"    # 192.168.3.1  (первый элемент)
echo "${ips[1]}"    # 192.168.3.109 (второй элемент)
echo "${ips[2]}"    # 192.168.3.116 (третий элемент)
echo "${ips[@]}"    # Вывести весь массив целиком

Обязательны фигурные скобки вокруг ips[0] , без фигурных скобок Bash поймет это неправильно
```

---

# Пример 1 
```bash
ips=("192.168.3.1" "192.168.3.109" "192.168.3.116")

for ip in "${ips[@]}"
do 
    ping -c1 $ip > /dev/null
    if [  $? -eq 0 ]; then 
        echo "$ip активен"
    else 
        echo "$ip неактивен"
    fi
done
```

# Пример 2
```bash
check_host() {
    ip=$1
    ping -c1 "$ip" > /dev/null
    if [ $? -eq 0 ]; then # проверяем код возврата
        echo "$ip - доступен"
        return 0
    else  
        echo "Обнаружен недоступный хост: $ip"
        return 1
    fi
        
}

protected=("192.168.3.109" "192.168.3.116" "192.168.3.1"  )   # Исключения
hosts_to_check=("192.168.3.109" "192.168.3.116" "192.168.3.1" "192.168.3.139" "8.8.8.8") # Хосты для проверки

for ip in "${hosts_to_check[@]}"  # перебираем массив по одному элементу за раз, кладём каждый в ip
do
    is_protected=0 # сбрасивыем флажок в 0, и считаем что хост не защищен, пока не доказано обратное
    for p in "${protected[@]}"  # перебираем защищённые IP по одному, кладём каждый в p
    do
        if [ "$ip" = "$p" ]; then  # сравнение двух отдельных строк (текущего элемента одного массива с текущим элементом другого), а не массивов целиком
            is_protected=1 # если есть в protected, то 1 
        fi
    done 
    if [ $is_protected -eq 1 ]; then  # проверяем если 1, то защищен
        echo "$ip защищен, пропускаю"
        continue
    fi

    check_host "$ip"
    if [ $? -ne 0 ]; then  # здесь если если хост НЕ ответил (недоступен) — пропускаем оставшийся код для него и переходим к следующему
        continue
    fi
done
```
