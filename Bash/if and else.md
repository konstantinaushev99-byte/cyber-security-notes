# if
Что это - это первая тема, где скрипт начинает сам принимать решения, а не просто выполнять команды одну за другой

---

# Синтаксис 
```bash
if [ условие ]; then  # [ условие ] c пробелами 
    команда
fi # Конец блока
```

---

# Примеры if
```bash
#!/bin/bash

count=5 
if [ $count -eq 5 ]; then # Условие ( если count = 5 )
    echo "count равно 5"  # То будет такой вывод
fi # Конец
```

```bash
#!bin/bash

ping -c1 192.168.3.131 > /dev/null  # Делаем пинг 1 раз, и прячем вывод пинга 

if [ $? -eq 0 ]; then # Выполнится если код возврата равен 0
  echo "Хост доступен"
fi
```

# Else 
Что это - что делать, если условие не выполнилось

---

# Примеры
```bash
#!/bin/bash

ping -c1 192.168.3.131 > /dev/null
if [ $? -eq 0 ]; then
    echo "Хост доступен"
else # если не 0, то хост недоступен или если хоста не существует 
    echo "Хост недоступен"
fi
```

```bash
Пример с аргументами и ping
#!/bin/bash

ip=$1 # аргумент $1

ping -c1 $ip > /dev/null
if [ $? -eq 0 ]; then
    echo "Хост доступен"
else
    echo "Хост недоступен"
fi
```

```bash
Пример с использованием ping и xargs
#!/bin/bash

ip=$1

ping -c1 $ip > /dev/null
if [ $? -eq 0 ]; then
    echo "Хост $ip доступен" # добавил $ip чтобы лучше читать вывод
else
    echo "Хост $ip недоступен"
fi

В терминале ведем: cat hosts.txt | xargs -n1 ./if.sh
```

# -z 
Что это  - нужен для одной конкретной проверки: "эта переменная вообще пустая, или в ней что-то есть?"

# Пример
```bash
#!/bin/bash

ip=$1

if [ -z "$ip" ]; then
    echo "Ошибка: укажите IP-адрес"
else
    ping -c1 $ip > /dev/null
    if [ $? -eq 0 ]; then
       echo "Хост $ip доступен"
    else
        echo "Хост $ip недоступен"
    fi
fi
```
