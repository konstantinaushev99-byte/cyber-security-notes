# Функции решают проблему повторения кода — вместо того чтобы копировать один и тот же блок несколько раз (как  было с проверкой ping в нескольких скриптах), можно написать его один раз и потом просто вызывать по имени

## Базовый синтаксис
```bash
function_name() {
    команды
}
```

### Слово function перед именем можно писать, а можно и нет — оба варианта работают одинаково:
```bash
function greet() {
    echo "Привет!"
}
```
или
```bash
greet() {
    echo "Привет!"
}
```

### Главное: функцию нужно сначала объявить, а потом уже вызвать, иначе bash её ещё не будет "знать":
```bash
#!/bin/bash

greet() {
    echo "Привет!"
}

greet    # вызов — просто пишем имя функции, без скобок при вызове
```

# Пример 1 ( с аргументами, здесь функция не так уж и нужна, но это просто наглядный пример )
```bash
#!/bin/bash

check_host() {
    ip=$1
    ping -c1 "$ip" > /dev/null
    if [ $? -eq 0 ]; then
        echo "$ip - доступен"
    else
        echo "$ip - недоступен"
    fi
}

check_host 192.168.3.109
check_host 192.168.3.1
check_host 8.8.8.8
```

# Пример 2 ( соединение функции с циклом )
```bash
check_host() {
    ip=$1
    ping -c1 "$ip" > /dev/null
    if [ $? -eq 0 ]; then
        echo "$ip - доступен"
    else
        echo "$ip - недоступен"
    fi
}

file=$1

for ip in $(cat "$file")
do
    check_host "$ip"
done

( если завтра захочу изменить, как именно проверяется хост (например, добавить проверку конкретного порта вместо просто ping), нужно будет поправить код только в одном месте (внутри функции), а не искать все места, где вы копировали этот блок ) 
```

# Пример 3 ( Return - код возврата функиции )
```bash
is_available() {
  ping -c1 "$1" > /dev/null
  return $?
}

if is_available 192.168.3.109; then
  echo "Доступен"
fi

( return $? — передаёт код возврата ping наружу, как будто это код возврата самой функции )
```

# Пример 4 ( прошлый код с break/countinue, но с функцией )
```bash
check_host() {
    ping -c1 "$ip" > /dev/null
    if [ $? -eq 0 ]; then
        echo "$ip - доступен"
        return 0
    else  
        echo "Обнаружен недоступный хост: $ip, прекращаю проверку"
        return 1
    fi
        
}

file=$1

for ip in $(cat "$file")
do 
    if [ "$ip" = "192.168.3.109" ]; then
        echo "$ip это моя виртуалка ее не трогать!!!"
        continue
    elif [ "$ip" = "192.168.3.116" ]; then
        echo "$ip это тоже моя виртуалка, не трогать"
        continue
    elif [ "$ip" = "192.168.3.1" ]; then
        echo "$ip это роутер, не трогать"
        continue
    fi

    check_host "$ip"
    if [ $? -ne 0 ]; then
        break
   fi
done
```

# Итог
- Функции окупаются не в маленьких скриптах, а в больших, когда есть несколько связанных, но разных этапов работы, и особенно когда одну и ту же логику нужно вызвать в нескольких местах. Для одноразовой простой проверки — можно и без них, это нормально
