# exit 
Что это - явно завершить скрипт с определённым кодом

---

# Пример 1 
```bash
#!/bin/bash

ip=$1

if [ -z "$ip" ]; then   # Проверка на пустую строку
    echo "Ошибка: укажите IP-адрес"
    exit 1
fi

echo "Проверяю $ip"

exit немедленно прерывает выполнение скрипта, ни одна строка после него в этом же скрипте не выполнится
```

# trap 
Что это - перехватить сигнал (например, Ctrl+C) и сделать что-то перед завершением

---

# Пример 1
### Допустим мой скрипт создаёт временный файл, а пользователь нажимает Ctrl+C посреди работы — временный файл так и останется, "мусором". trap позволяет перехватить такое прерывание и выполнить "уборку" перед реальным завершением
```bash
#!/bin/bash

trap 'echp "Скрипт прерван, убираю за собой..."; rm -f temp.txt; exit 1' INT

echo "test" > temp.txt
echo "Работаю... нажмите Ctrl+C для  проверки"
sleep 10
echo "Готово"
```
1. trap '...' INT - если получишь сигнал INT (это именно то, что посылает Ctrl+C), выполни вот этот код в кавычках вместо обычного немедленного завершения
2. Внутри кавычек — обычные команды через ;, как одна строка
3. Если во время sleep 10 нажать Ctrl+C, вместо мгновенного обрыва скрипт сначала выведет сообщение, удалит temp.txt, и только потом завершится

# Практический пример с exit и trap
```bash
trap 'echo "Скинирование прервано пользователем"; exit 1;' INT
echo "Для остановки нажмите - Ctrl+C"       # досрочно завершает выплнение скрипта                        
sleep 5 

check_host() {   # функиция
    ip=$1
    ping -c1 "$ip" > /dev/null
    if [ $? -eq 0 ]; then
        echo "$ip - доступен"
        return 0
    else  
        echo "Обнаружен недоступный хост: $ip, прекращаю проверку"
        return 1
    fi
        
}

file=$1

if [ -z "$file" ]; then       # Проверка на то что файл есть(проверка на пустую строку)
    echo "Укажите файл с хостами"      
    exit 1
fi

if [ ! -f "$file" ]; then   # Проверка что файл существует
    echo "Файл $file несуществует, укажите действующей файл"  
    exit 1
fi

while read -r ip   # Цикл while
do 
    ip=$(echo "$ip" | cut -d'#' -f1 | xargs) # задаем разделитель '#', и выводим первое поле (-f1) 

    if [ "$ip" = "192.168.3.109" ]; then
        echo "$ip это моя виртуалка ее не трогать!!!"
        continue
    elif [ "$ip" = "192.168.3.116" ]; then
        echo "$ip это тоже моя виртуалка, не трогать"
        continue
    elif [ "$ip" = "192.168.3.1" ]; then
        echo "$ip это роутер, не трогать"
        continue
    fi

    check_host "$ip"
    if [ $? -ne 0 ]; then   # обращение к функиции
        break
    fi
done < "$file"
